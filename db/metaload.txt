/*all loading instructions for filling the idf tables (qf tables are filled by c# program)*/


INSERT INTO numtuples
SELECT COUNT(*)
FROM autompg;


/*categorical attributes idf scores*/

/*brand*/
INSERT INTO brandidf SELECT brand, LOG(numtuples.num/COUNT(brand)) 
FROM autompg
INNER JOIN numtuples
GROUP BY brand

/*model*/
INSERT INTO modelidf SELECT model, LOG(numtuples.num/COUNT(model)) 
FROM autompg
INNER JOIN numtuples
GROUP BY model

/*type*/
INSERT INTO typeidf SELECT type, LOG(numtuples.num/COUNT(type)) 
FROM autompg
INNER JOIN numtuples
GROUP BY type

/*origin*/
INSERT INTO originidf SELECT origin, LOG(numtuples.num/COUNT(origin)) 
FROM autompg
INNER JOIN numtuples
GROUP BY origin



/*numerical attributes idf scores by formula (2) of the paper*/

/*mpg*/
/*h-value only temporary table*/
CREATE TABLE mpgbandwidth(
    bandwidth real
);

INSERT INTO mpgbandwidth 
SELECT 1.06*STDEV(mpg)*POWER(numtuples.num, -0.2) AS bandwidth 
FROM autompg
INNER JOIN numtuples;

/*see formula (2) of paper*/
INSERT INTO mpgidf
SELECT mpg1, LOG10(numtuples.num/summed) FROM (
SELECT mpg1, SUM( EXP(-0.5 * POWER( (mpg2-mpg1)/mpgbandwidth.bandwidth , 2 ) ) ) AS summed
FROM (SELECT DISTINCT mpg AS mpg1 FROM autompg), (SELECT mpg AS mpg2 FROM autompg)
INNER JOIN mpgbandwidth
GROUP BY mpg1)
INNER JOIN numtuples;

DROP TABLE mpgbandwidth;
